# The JSONS data has been stored in the data variable.
# [Post(title=p["title"], content=p["content"], author_id=p["user_id"]) for p in data] : this is the iteration  via for loop .


import json
from blog.models import Post

with open("posts.json", encoding="utf-8") as f:
    data = json.load(f)

Post.objects.bulk_create(
    [Post(title=p["title"], content=p["content"], author_id=p["user_id"]) for p in data]
)
print(Post.objects.count())
